import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import com.taskcheck.vikrayon.model.VikrayaUser;
import com.taskcheck.vikrayon.repository.UserRepository;



@RestController
@RequestMapping("/home")
public class UserController {
	
	@Autowired
	private UserRepository userRepository;
		
	@PostMapping("/register")
	public ResponseEntity<String> postUser(@RequestBody VikrayaUser user){
		userRepository.save(user);
		System.out.println(user.toString());
		return ResponseEntity.ok("registered");
	}
	
	@GetMapping("/register")
	public List<VikrayaUser> getUser(){
		System.out.println(userRepository.findAll() + "checking");
		return userRepository.findAll();
	}
}

package com.taskcheck.vikrayon.model;


import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.SequenceGenerator;
import lombok.AllArgsConstructor;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;
import lombok.ToString;


@NoArgsConstructor
@AllArgsConstructor
@ToString
@Entity
public class VikrayaUser {

	 private static final String Confirmpassword = "12345";

	private static final String Forgetpassword = "12345";

	@Id
	 @GeneratedValue(strategy = GenerationType.SEQUENCE, generator = "vikraya_seq")
	    @SequenceGenerator(name = "vikraya_seq", sequenceName = "vikraya_seq", allocationSize = 1)
	    private Long id;
	 
	    private String Username;
	    private String Email;
	    private String Password;
	 	 private String Phone;
	  
	 
	    @Enumerated(EnumType.STRING)
	    private Role role;

		public Long getId() {
			return id;
		}

		public void setId(Long id) {
			this.id = id;
		}

		public String getUsername() {
			return Username;
		}

		public void setUsername(String username) {
			Username = username;
		}

		public String getEmail() {
			return Email;
		}

		public void setEmail(String email) {
			Email = email;
		}

		public String getPassword() {
			return Password;
		}

		public void setPassword(String password) {
			Password = password;
		}
		public String getConfirmpassword() {
			return Confirmpassword;
		}
		public void setConfirmpassword( String Confirmpassword) {
			 Confirmpassword = Confirmpassword;
		}

		public String getPhone() {
			return Phone;
		}

		public void setPhone(String phone) {
			Phone = phone;
		}

		public Role getSignup() {
			return role;
		}

		public void setsignup(Role Signup) {
			this.role = Signup;
		}
		public String setForgetpassword() {
		return  Forgetpassword;
}
          
		

		@Override
		public String toString() {
			return "VikrayaUser [id=" + id + ", Username=" + Username + ", Email=" + Email + ", Password=" + Password
					+ ", Phone=" + Phone + ", role=" + role + "]";
		}
	    
	    
	    
}
enum Role {
    USER,
    VENDOR,
}
